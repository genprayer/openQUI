cmake_minimum_required(VERSION 3.24)
project(OpenQUI VERSION 0.6.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_AUTOMOC ON)

include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Qml DBus Concurrent Svg Xml LinguistTools)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
qt_standard_project_setup(REQUIRES 6.8)

add_library(open_icue_core STATIC
    src/core/workspace.cpp src/core/workspace.h
    src/devices/catalog.cpp src/devices/catalog.h
    src/devices/inventory.cpp src/devices/inventory.h
    src/devices/hid_transport.cpp src/devices/hid_transport.h
    src/devices/hardware_backend.cpp src/devices/hardware_backend.h
    src/devices/keyboard_input.cpp src/devices/keyboard_input.h
    src/profiles/cereal.cpp src/profiles/cereal.h
    src/audio/filter_chain.cpp src/audio/filter_chain.h
    src/audio/alsa_mixer.cpp src/audio/alsa_mixer.h
    src/audio/routing.cpp src/audio/routing.h
)
target_include_directories(open_icue_core PUBLIC src)
target_link_libraries(open_icue_core PUBLIC Qt6::Core Qt6::Gui Qt6::Xml PkgConfig::UDEV PkgConfig::ALSA)
target_compile_options(open_icue_core PRIVATE -Wall -Wextra -Wpedantic)

set(OPENQUI_USE_LOCAL_CORSAIR_RESOURCES "AUTO" CACHE STRING "Original iCUE resources: AUTO, ON or OFF")
set_property(CACHE OPENQUI_USE_LOCAL_CORSAIR_RESOURCES PROPERTY STRINGS AUTO ON OFF)
string(TOUPPER "${OPENQUI_USE_LOCAL_CORSAIR_RESOURCES}" LOCAL_RESOURCE_MODE)
if(NOT LOCAL_RESOURCE_MODE MATCHES "^(AUTO|ON|OFF)$")
    message(FATAL_ERROR "OPENQUI_USE_LOCAL_CORSAIR_RESOURCES must be AUTO, ON or OFF")
endif()
set(REQUIRED_DEVICE_BUNDLES darkstar_wireless k70_core_rgb virtuoso_black virtuoso_black_dongle slipstream_wireless_dongle_ss)
set(MISSING_DEVICE_BUNDLES "")
foreach(MODEL_ID IN LISTS REQUIRED_DEVICE_BUNDLES)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resources/vendor/${MODEL_ID}.rcc" AND
       NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resources/vendor/library/bundles/${MODEL_ID}.rcc")
        list(APPEND MISSING_DEVICE_BUNDLES ${MODEL_ID})
    endif()
endforeach()
set(IS_USING_LOCAL_CORSAIR_RESOURCES OFF)
if(LOCAL_RESOURCE_MODE STREQUAL "ON")
    if(MISSING_DEVICE_BUNDLES)
        message(FATAL_ERROR "Original resources requested, but device bundles are missing: ${MISSING_DEVICE_BUNDLES}")
    endif()
    set(IS_USING_LOCAL_CORSAIR_RESOURCES ON)
elseif(LOCAL_RESOURCE_MODE STREQUAL "AUTO" AND NOT MISSING_DEVICE_BUNDLES)
    set(IS_USING_LOCAL_CORSAIR_RESOURCES ON)
endif()
message(STATUS "OpenQUI original device resources: ${IS_USING_LOCAL_CORSAIR_RESOURCES} (requested ${LOCAL_RESOURCE_MODE})")
if(IS_USING_LOCAL_CORSAIR_RESOURCES)
    file(GLOB VENDOR_BUNDLES CONFIGURE_DEPENDS resources/vendor/*.rcc resources/vendor/library/bundles/*.rcc)
    file(GLOB_RECURSE VENDOR_ARTWORK CONFIGURE_DEPENDS resources/vendor/fonts/* resources/vendor/cue-icons/* resources/vendor/cue-assets/* resources/vendor/voice/*)
    set(VENDOR_RESOURCES ${VENDOR_BUNDLES} ${VENDOR_ARTWORK})
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resources/vendor/equalizer-presets.xml")
        list(APPEND VENDOR_RESOURCES resources/vendor/equalizer-presets.xml)
    endif()
    if(NOT VENDOR_BUNDLES)
        message(FATAL_ERROR "Local resources requested, but no RCC bundles were imported")
    endif()
    qt_add_resources(open_icue_core original_device_resources BIG_RESOURCES PREFIX /vendor BASE resources/vendor FILES ${VENDOR_RESOURCES})
    file(GLOB_RECURSE IMPORTED_RUNTIME_RESOURCES CONFIGURE_DEPENDS resources/vendor/library/runtime/*)
    if(IMPORTED_RUNTIME_RESOURCES)
        qt_add_resources(open_icue_core imported_runtime_resources BIG_RESOURCES PREFIX /vendor BASE resources/vendor/library/runtime FILES ${IMPORTED_RUNTIME_RESOURCES})
    endif()
endif()
target_compile_definitions(open_icue_core PRIVATE OPENQUI_LOCAL_CORSAIR_RESOURCES=$<BOOL:${IS_USING_LOCAL_CORSAIR_RESOURCES}>)
file(GLOB_RECURSE APPLICATION_RESOURCES CONFIGURE_DEPENDS resources/keyboard-layouts/*.json resources/icons/*.svg resources/icons/*.json resources/branding/*.png resources/branding/*.svg resources/devices/*.json resources/devices/*.svg resources/audio/*.wav resources/audio/*.md)
qt_add_resources(open_icue_core application_resources PREFIX /app BASE resources FILES ${APPLICATION_RESOURCES})

add_library(open_icue_controller STATIC src/gui/application_model.cpp src/gui/application_model.h src/gui/translation_manager.cpp src/gui/translation_manager.h)
target_link_libraries(open_icue_controller PUBLIC open_icue_core)
target_link_libraries(open_icue_controller PRIVATE Qt6::Qml Qt6::Concurrent)
target_compile_options(open_icue_controller PRIVATE -Wall -Wextra -Wpedantic)
qt_add_lrelease(TS_FILES translations/open-icue_de.ts translations/open-icue_en.ts translations/open-icue_ru.ts QM_FILES_OUTPUT_VARIABLE UI_QM_FILES)
qt_add_resources(open_icue_controller ui_translations PREFIX /i18n BASE ${CMAKE_CURRENT_BINARY_DIR} FILES ${UI_QM_FILES})
set(DEVICE_VIEW_SOURCES src/gui/device_images.cpp src/gui/device_images.h src/gui/device_selection.cpp src/gui/device_selection.h src/gui/icon_images.cpp src/gui/icon_images.h)
qt_add_executable(open-cue-gui src/main.cpp ${DEVICE_VIEW_SOURCES} src/gui/system_tray.cpp src/gui/system_tray.h src/service/dbus_adaptor.cpp src/service/dbus_adaptor.h)
target_link_libraries(open-cue-gui PRIVATE open_icue_controller Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::DBus Qt6::Concurrent Qt6::Svg)
target_compile_options(open-cue-gui PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(open-cue-gui PRIVATE OPENQUI_VERSION="${PROJECT_VERSION}")
file(GLOB_RECURSE QML_SOURCES CONFIGURE_DEPENDS qml/*.qml qml/qmldir)
qt_add_resources(open-cue-gui interface_resources PREFIX / BASE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${QML_SOURCES})
